    <script>
        const DEFAULT_SUBJECTS = [
            { id: 1, name: "General Medicine - Theory", category: "Theory", attended: 22, total: 28, target: 75, history: [] },
            { id: 2, name: "General Surgery - Theory", category: "Theory", attended: 19, total: 26, target: 75, history: [] },
            { id: 3, name: "Obstetrics & Gynecology", category: "Theory", attended: 24, total: 28, target: 80, history: [] },
            { id: 4, name: "Pediatrics", category: "Theory", attended: 15, total: 22, target: 75, history: [] },
            { id: 5, name: "Orthopedics", category: "Theory", attended: 12, total: 18, target: 75, history: [] },
            { id: 6, name: "Ophthalmology", category: "Theory", attended: 14, total: 16, target: 75, history: [] },
            { id: 7, name: "Otolaryngology (ENT)", category: "Theory", attended: 13, total: 18, target: 75, history: [] },
            { id: 8, name: "Community Medicine (PSM)", category: "Theory", attended: 25, total: 30, target: 80, history: [] },
            { id: 9, name: "Pharmacology", category: "Theory", attended: 30, total: 35, target: 75, history: [] },
            { id: 10, name: "Pathology", category: "Theory", attended: 28, total: 32, target: 75, history: [] },
            { id: 11, name: "Medicine Ward Posting", category: "Posting", attended: 18, total: 20, target: 85, history: [] },
            { id: 12, name: "Surgery Ward Posting", category: "Posting", attended: 16, total: 20, target: 85, history: [] },
            { id: 13, name: "OBG Clinicals", category: "Practical", attended: 14, total: 15, target: 80, history: [] },
            { id: 14, name: "Pediatrics Clinicals", category: "Practical", attended: 12, total: 15, target: 80, history: [] },
            { id: 15, name: "Community Medicine Field Visit", category: "Practical", attended: 10, total: 12, target: 75, history: [] }
        ];

        let subjects = JSON.parse(localStorage.getItem('oled_attendance_subjects')) || DEFAULT_SUBJECTS;
        let currentFilter = 'All';

        function saveData() {
            localStorage.setItem('oled_attendance_subjects', JSON.stringify(subjects));
            renderApp();
        }

        function setFilter(cat, el) {
            currentFilter = cat;
            document.querySelectorAll('.filter-chip').forEach(c => c.classList.remove('active'));
            el.classList.add('active');
            renderApp();
        }

        function markAttendance(id, present) {
            const sub = subjects.find(s => s.id === id);
            if (!sub) return;
            sub.history.push({ attended: sub.attended, total: sub.total });
            sub.total += 1;
            if (present) sub.attended += 1;
            saveData();
        }

        function undoAttendance(id) {
            const sub = subjects.find(s => s.id === id);
            if (!sub || sub.history.length === 0) return;
            const last = sub.history.pop();
            sub.attended = last.attended;
            sub.total = last.total;
            saveData();
        }

        function updateTarget(id) {
            const sub = subjects.find(s => s.id === id);
            if (!sub) return;
            const newT = prompt(`Enter new target percentage for ${sub.name}:`, sub.target);
            if (newT !== null && !isNaN(newT) && newT > 0 && newT <= 100) {
                sub.target = parseInt(newT);
                saveData();
            }
        }

        function renderApp() {
            const listEl = document.getElementById('subjectList');
            listEl.innerHTML = '';

            const filtered = currentFilter === 'All' ? subjects : subjects.filter(s => s.category === currentFilter);

            if (filtered.length === 0) {
                listEl.innerHTML = `<div style="text-align:center; color: var(--text-secondary); padding: 40px;">No subjects found.</div>`;
                return;
            }

            filtered.forEach(sub => {
                const pct = sub.total === 0 ? 0 : Math.round((sub.attended / sub.total) * 100);
                const isSafe = pct >= sub.target;
                const color = isSafe ? 'var(--accent-green)' : 'var(--accent-red)';
                
                // Donut math
                const radius = 28;
                const circ = 2 * Math.PI * radius;
                const offset = circ - (pct / 100) * circ;

                // Smart margin calculation
                let marginHtml = '';
                if (sub.total === 0) {
                    marginHtml = `<span style="color: var(--text-secondary)">No classes logged yet</span>`;
                } else if (isSafe) {
                    // Safe to miss: how many can we miss while keeping pct >= target?
                    // (attended) / (total + x) >= target/100
                    let x = 0;
                    while ((sub.attended / (sub.total + x + 1)) >= (sub.target / 100)) {
                        x++;
                    }
                    if (x > 0) {
                        marginHtml = `<span style="color: var(--accent-green)">Can safely miss ${x} class${x > 1 ? 'es' : ''}</span>`;
                    } else {
                        marginHtml = `<span style="color: var(--accent-yellow, #FFD60A)">On threshold limit</span>`;
                    }
                } else {
                    // Need to attend: how many consecutive classes needed?
                    // (attended + y) / (total + y) >= target/100
                    let y = 0;
                    let currAtt = sub.attended;
                    let currTot = sub.total;
                    while ((currAtt / currTot) < (sub.target / 100) && y < 500) {
                        currAtt++;
                        currTot++;
                        y++;
                    }
                    marginHtml = `<span style="color: var(--accent-red)">Attend next ${y} class${y > 1 ? 'es' : ''}</span>`;
                }

                const card = document.createElement('div');
                card.className = 'subject-card';
                card.style.borderColor = isSafe ? 'rgba(48,209,88,0.2)' : 'rgba(255,69,58,0.3)';
                
                card.innerHTML = `
                    <div class="card-header">
                        <div>
                            <div class="subject-title">${sub.name}</div>
                            <div class="category-tag">${sub.category}</div>
                        </div>
                        <button class="target-edit" onclick="updateTarget(${sub.id})">Target: ${sub.target}%</button>
                    </div>
                    <div class="card-body">
                        <div class="stats-col">
                            <div class="stat-text">Attended: <span class="stat-val">${sub.attended} / ${sub.total}</span></div>
                            <div class="margin-text">${marginHtml}</div>
                        </div>
                        <div class="donut-container">
                            <svg class="donut-svg" viewBox="0 0 72 72">
                                <circle class="donut-bg" cx="36" cy="36" r="${radius}"></circle>
                                <circle class="donut-fill" cx="36" cy="36" r="${radius}" 
                                    style="stroke: ${color}; stroke-dasharray: ${circ}; stroke-dashoffset: ${offset};">
                                </circle>
                            </svg>
                            <div class="donut-center" style="color: ${color};">${pct}%</div>
                        </div>
                    </div>
                    <div class="actions-row">
                        <button class="btn-yes" onclick="markAttendance(${sub.id}, true)">Yes (Present)</button>
                        <button class="btn-no" onclick="markAttendance(${sub.id}, false)">No (Absent)</button>
                        ${sub.history.length > 0 ? `<button class="btn-undo" onclick="undoAttendance(${sub.id})">Undo</button>` : ''}
                    </div>
                `;
                listEl.appendChild(card);
            });
        }

        function openAddModal() { document.getElementById('addModal').classList.add('active'); }
        function closeAddModal() { document.getElementById('addModal').classList.remove('active'); }

        function saveNewSubject() {
            const name = document.getElementById('newSubName').value.trim();
            const category = document.getElementById('newSubCat').value;
            const target = parseInt(document.getElementById('newSubTarget').value) || 75;
            const attended = parseInt(document.getElementById('newSubAttended').value) || 0;
            const total = parseInt(document.getElementById('newSubTotal').value) || 0;

            if (!name) { alert('Please enter a subject name'); return; }
            if (attended > total) { alert('Attended cannot be greater than total classes'); return; }

            subjects.push({
                id: Date.now(),
                name,
                category,
                attended,
                total: Math.max(attended, total),
                target,
                history: []
            });

            closeAddModal();
            document.getElementById('newSubName').value = '';
            document.getElementById('newSubAttended').value = '0';
            document.getElementById('newSubTotal').value = '0';
            saveData();
        }

        function resetAllData() {
            if (confirm('Reset all attendance data back to default subjects?')) {
                subjects = JSON.parse(JSON.stringify(DEFAULT_SUBJECTS));
                saveData();
            }
        }

        renderApp();
    </script>
